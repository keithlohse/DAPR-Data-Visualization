2 Quality Assurance & Feature Extraction

From raw Center of Pressure (COP) time-series data \((COP_x, COP_y)\), we extract key biomechanical features:

  1. Path Lengths: Sum of absolute step distances across axes: \[\text{PathLength}_{\text{AP}} = \sum_{n=1}^{N-1} |COP_{y, n+1} - COP_{y, n}|\] \[\text{PathLength}_{\text{Total}} = \sum_{n=1}^{N-1} \sqrt{(COP_{x, n+1} - COP_{x, n})^2 + (COP_{y, n+1} - COP_{y, n})^2}\]

  2. 95% Confidence Ellipse Area & Boundary: \[\text{Area}_{95} = 2 \pi F_{0.95, 2, N-2} \sqrt{\lambda_1 \lambda_2}\] where \(\lambda_1, \lambda_2\) are eigenvalues of the \((COP_x, COP_y)\) covariance matrix.

  • Python (numpy, scipy & matplotlib)

# 1. Feature Extraction & Ellipse Math
compute_path_metrics <- function(x, y, duration_s) {
  dx <- diff(x)
  dy <- diff(y)
  pl_ml    <- sum(abs(dx))
  pl_ap    <- sum(abs(dy))
  pl_total <- sum(sqrt(dx^2 + dy^2))
  
  tibble(
    PathLength_AP    = pl_ap,
    PathLength_ML    = pl_ml,
    PathLength_Total = pl_total,
    MeanVelocity_AP  = pl_ap / duration_s,
    MeanVelocity_ML  = pl_ml / duration_s
  )
}

compute_ellipse95 <- function(x, y) {
  n <- length(x)
  cov_mat <- stats::cov(cbind(x, y))
  eig <- eigen(cov_mat)
  Fval <- stats::qf(0.95, df1 = 2, df2 = n - 2)
  area <- 2 * pi * Fval * sqrt(eig$values[1] * eig$values[2])
  
  list(area = area, eig = eig, Fval = Fval, center = c(mean(x), mean(y)))
}

ellipse_boundary_df <- function(x, y, n_points = 100) {
  e <- compute_ellipse95(x, y)
  axis_lengths <- sqrt(2 * e$Fval * e$eig$values)
  theta  <- seq(0, 2 * pi, length.out = n_points)
  circle <- rbind(cos(theta), sin(theta))
  pts <- e$eig$vectors %*% diag(axis_lengths) %*% circle
  
  tibble(ex = e$center[1] + pts[1, ], ey = e$center[2] + pts[2, ])
}

compute_trial_features <- function(df) {
  n        <- nrow(df)
  duration <- max(df$Time) - min(df$Time)
  path_m   <- compute_path_metrics(df$COPx, df$COPy, duration)
  ellipse  <- compute_ellipse95(df$COPx, df$COPy)
  time_d   <- diff(df$Time)
  
  bind_cols(
    path_m,
    tibble(
      EllipseArea95     = ellipse$area,
      n_samples         = n,
      duration_s        = duration,
      Fz_min            = min(df$Fz),
      Fz_median         = median(df$Fz),
      time_gap_max      = max(time_d),
      time_nonmonotonic = any(time_d <= 0)
    )
  )
}

make_subject_qa_plot <- function(subject_id, df) {
  p_top <- ggplot(df, aes(x = Time, y = CumulativePath, group = Trial, color = Rep, linetype = Vision)) +
    geom_line(linewidth = 0.4) +
    facet_wrap(~ Surface) +
    labs(title = paste("Subject", subject_id, "- QA Diagnostic"),
         subtitle = "Cumulative COP Path Length (slope ~ velocity; jumps -> possible artifact)",
         x = "Time (s)", y = "Cumulative path (cm)") +
    theme_minimal(base_size = 9)
  
  ellipses <- df %>%
    group_by(Trial, Vision, Surface, Rep) %>%
    group_modify(~ ellipse_boundary_df(.x$COPx, .x$COPy)) %>%
    ungroup()
  
  p_bottom <- ggplot(df, aes(x = COPx, y = COPy, group = Trial, color = Rep, linetype = Vision)) +
    geom_path(linewidth = 0.3, alpha = 0.5) +
    geom_path(data = ellipses, aes(x = ex, y = ey, group = Trial), linewidth = 0.5, alpha = 0.9) +
    facet_wrap(~ Surface) +
    coord_equal() +
    labs(subtitle = "Stabilogram with 95% Confidence Ellipses", x = "COPx (cm)", y = "COPy (cm)") +
    theme_minimal(base_size = 9)
  
  (p_top / p_bottom) + plot_layout(guides = "collect")
}

import os

from scipy.stats import f
import matplotlib.pyplot as plt
import seaborn as sns

master = pd.read_parquet("./data/BDS/output/BDS_master.parquet")

# 1. Feature Extraction & Ellipse Math
def compute_ellipse95(x, y):
    n = len(x)
    cov_mat = np.cov(x, y)
    eigvals, eigvecs = np.linalg.eigh(cov_mat)
    f_val = f.ppf(0.95, 2, n - 2)
    area = 2 * np.pi * f_val * np.sqrt(np.maximum(0, eigvals[0] * eigvals[1]))
    return area, eigvals, eigvecs, f_val, np.mean(x), np.mean(y)

def ellipse_boundary_df(x, y, n_points=100):
    area, eigvals, eigvecs, f_val, cx, cy = compute_ellipse95(x, y)
    axis_lengths = np.sqrt(2 * f_val * np.maximum(0, eigvals))
    theta = np.linspace(0, 2 * np.pi, n_points)
    circle = np.vstack([np.cos(theta), np.sin(theta)])
    pts = eigvecs @ np.diag(axis_lengths) @ circle
    return pd.DataFrame({'ex': cx + pts[0, :], 'ey': cy + pts[1, :]})

def compute_trial_features(df):
    duration = df["Time"].max() - df["Time"].min()
    dx = np.diff(df["COPx"])
    dy = np.diff(df["COPy"])
    
    pl_ml  = np.sum(np.abs(dx))
    pl_ap  = np.sum(np.abs(dy))
    pl_tot = np.sum(np.sqrt(dx**2 + dy**2))
    
    area, _, _, _, _, _ = compute_ellipse95(df["COPx"], df["COPy"])
    time_diffs = np.diff(df["Time"])
    
    return pd.Series({
        "PathLength_AP": pl_ap, "PathLength_ML": pl_ml, "PathLength_Total": pl_tot,
        "MeanVelocity_AP": pl_ap / duration, "MeanVelocity_ML": pl_ml / duration,
        "EllipseArea95": area, "n_samples": len(df), "duration_s": duration,
        "Fz_min": df["Fz"].min(), "Fz_median": df["Fz"].median(),
        "time_gap_max": time_diffs.max(), "time_nonmonotonic": (time_diffs <= 0).any()
    })

meta_cols = ["Trial", "Subject", "Vision", "Surface", "Age", "AgeGroup", "Gender", "Height", "Weight", "BMI"]
trial_summary = master.groupby(meta_cols, observed=True).apply(compute_trial_features).reset_index()

def make_subject_qa_plot(subject_id, df_sub):
    fig, axes = plt.subplots(2, 2, figsize=(9, 8), sharex='row', sharey='row')
    palette = {1: "#1E6654", 2: "#CC5500", 3: "#5B7C99"}
    surfaces = ["Firm", "Foam"]
    
    for j, surf in enumerate(surfaces):
        df_surf = df_sub[df_sub["Surface"] == surf]
        ax_top = axes[0, j]

        ax_top.set_xlabel("Time (s)")

# Quality Assurance & Feature Extraction

From raw Center of Pressure (COP) time-series data $(COP_x, COP_y)$, we extract key biomechanical features:

1. **Path Lengths**: Sum of absolute step distances across axes:
   $$\text{PathLength}_{\text{AP}} = \sum_{n=1}^{N-1} |COP_{y, n+1} - COP_{y, n}|$$
   $$\text{PathLength}_{\text{Total}} = \sum_{n=1}^{N-1} \sqrt{(COP_{x, n+1} - COP_{x, n})^2 + (COP_{y, n+1} - COP_{y, n})^2}$$

2. **95% Confidence Ellipse Area & Boundary**:
   $$\text{Area}_{95} = 2 \pi F_{0.95, 2, N-2} \sqrt{\lambda_1 \lambda_2}$$
   where $\lambda_1, \lambda_2$ are eigenvalues of the $(COP_x, COP_y)$ covariance matrix.

library(arrow)

master <- arrow::read_parquet("./data/BDS/output/BDS_master.parquet")

# 1. Feature Extraction & Ellipse Math
compute_path_metrics <- function(x, y, duration_s) {
  dx <- diff(x)
  dy <- diff(y)
  pl_ml    <- sum(abs(dx))
  pl_ap    <- sum(abs(dy))
  pl_total <- sum(sqrt(dx^2 + dy^2))
  
  tibble(
    PathLength_AP    = pl_ap,
    PathLength_ML    = pl_ml,
    PathLength_Total = pl_total,
    MeanVelocity_AP  = pl_ap / duration_s,
    MeanVelocity_ML  = pl_ml / duration_s
  )
}

compute_ellipse95 <- function(x, y) {
  n <- length(x)
  cov_mat <- stats::cov(cbind(x, y))
  eig <- eigen(cov_mat)
  Fval <- stats::qf(0.95, df1 = 2, df2 = n - 2)
  area <- 2 * pi * Fval * sqrt(eig$values[1] * eig$values[2])
  
  list(area = area, eig = eig, Fval = Fval, center = c(mean(x), mean(y)))
}

ellipse_boundary_df <- function(x, y, n_points = 100) {
  e <- compute_ellipse95(x, y)
  axis_lengths <- sqrt(2 * e$Fval * e$eig$values)
  theta  <- seq(0, 2 * pi, length.out = n_points)
  circle <- rbind(cos(theta), sin(theta))
  pts <- e$eig$vectors %*% diag(axis_lengths) %*% circle
  
  tibble(ex = e$center[1] + pts[1, ], ey = e$center[2] + pts[2, ])
}

compute_trial_features <- function(df) {
  n        <- nrow(df)
  duration <- max(df$Time) - min(df$Time)
  path_m   <- compute_path_metrics(df$COPx, df$COPy, duration)
  ellipse  <- compute_ellipse95(df$COPx, df$COPy)
  time_d   <- diff(df$Time)
  
  bind_cols(
    path_m,
    tibble(
      EllipseArea95     = ellipse$area,
      n_samples         = n,
      duration_s        = duration,
      Fz_min            = min(df$Fz),
      Fz_median         = median(df$Fz),
      time_gap_max      = max(time_d),
      time_nonmonotonic = any(time_d <= 0)
    )
  )
}

import os
import numpy as np
import pandas as pd
from scipy.stats import f
import matplotlib.pyplot as plt
import seaborn as sns

master = pd.read_parquet("./data/BDS/output/BDS_master.parquet")

# 1. Feature Extraction & Ellipse Math
def compute_ellipse95(x, y):
    n = len(x)
    cov_mat = np.cov(x, y)
    eigvals, eigvecs = np.linalg.eigh(cov_mat)
    f_val = f.ppf(0.95, 2, n - 2)
    area = 2 * np.pi * f_val * np.sqrt(np.maximum(0, eigvals[0] * eigvals[1]))
    return area, eigvals, eigvecs, f_val, np.mean(x), np.mean(y)

def ellipse_boundary_df(x, y, n_points=100):
    area, eigvals, eigvecs, f_val, cx, cy = compute_ellipse95(x, y)
    axis_lengths = np.sqrt(2 * f_val * np.maximum(0, eigvals))
    theta = np.linspace(0, 2 * np.pi, n_points)
    circle = np.vstack([np.cos(theta), np.sin(theta)])
    pts = eigvecs @ np.diag(axis_lengths) @ circle
    return pd.DataFrame({'ex': cx + pts[0, :], 'ey': cy + pts[1, :]})

def compute_trial_features(df):
    duration = df["Time"].max() - df["Time"].min()
    dx = np.diff(df["COPx"])
    dy = np.diff(df["COPy"])
    
    pl_ml  = np.sum(np.abs(dx))
    pl_ap  = np.sum(np.abs(dy))
    pl_tot = np.sum(np.sqrt(dx**2 + dy**2))
    
    area, _, _, _, _, _ = compute_ellipse95(df["COPx"], df["COPy"])
    time_diffs = np.diff(df["Time"])
    
    return pd.Series({
        "PathLength_AP": pl_ap, "PathLength_ML": pl_ml, "PathLength_Total": pl_tot,
        "MeanVelocity_AP": pl_ap / duration, "MeanVelocity_ML": pl_ml / duration,
        "EllipseArea95": area, "n_samples": len(df), "duration_s": duration,
        "Fz_min": df["Fz"].min(), "Fz_median": df["Fz"].median(),
        "time_gap_max": time_diffs.max(), "time_nonmonotonic": (time_diffs <= 0).any()
    })

meta_cols = ["Trial", "Subject", "Vision", "Surface", "Age", "AgeGroup", "Gender", "Height", "Weight", "BMI"]
trial_summary = master.groupby(meta_cols, observed=True).apply(compute_trial_features).reset_index()

trial_summary["flag_any"]            = trial_summary["flag_sample_count"] | trial_summary["flag_time_gap"] | trial_summary["flag_fz_unweighting"]
